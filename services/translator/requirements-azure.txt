# Dependencies for TRANSLATOR_ENGINE=azure.
#
# No torch, no fairseq2, no seamless_communication, no CUDA, no model
# weights: Azure does the translation and the synthesis, so this service
# is reduced to audio plumbing between LiveKit and Azure. That is why it
# runs on a lab server with no GPU at all.
#
#   python -m pip install -r requirements-azure.txt
#   TRANSLATOR_ENGINE=azure python -m uvicorn translator.app:app --port 8080

# LiveKit's realtime SDK: joins the class room, subscribes to the
# teacher's mic and publishes one audio track per target language.

numpy>=1.26,<3.0
soxr>=0.5
soundfile>=0.12
